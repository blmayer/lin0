+#!/bin/sh
+# hotplugd — minimal mdev-style hotplug helper for lin0.
+#
+# Modes:
+#   hotplugd -s     coldplug once (fix devices already present)
+#   hotplugd -d     background poll loop (works without CONFIG_UEVENT_HELPER)
+#   hotplugd        kernel usermode helper: ACTION/SUBSYSTEM/DEVNAME in env
+#
+# Register the helper (needs CONFIG_UEVENT_HELPER=y):
+#   echo /bin/hotplugd > /proc/sys/kernel/hotplug
+#
+# Rules are hardcoded for now (input nodes → root:input 0660).
+
+export PATH=/bin
+
+# --- rules (hardcoded; make config-driven later) ----------------------------
+
+# /dev/input/* and /dev/uinput → group input (gid 45), mode 0660
+rule_input() {
+	_node=$1
+	[ -e "$_node" ] || return 0
+	case "$_node" in
+	/dev/input/*|/dev/uinput)
+		chown root:input "$_node" 2>/dev/null
+		chmod 660 "$_node" 2>/dev/null
+		;;
+	esac
+}
+
+apply_input_all() {
+	for _n in /dev/input/* /dev/uinput; do
+		rule_input "$_n"
+	done
+}
+
+# One kernel uevent (usermode helper path).
+handle_uevent() {
+	case "${ACTION:-}" in
+	add|change) ;;
+	*) return 0 ;;
+	esac
+
+	case "${SUBSYSTEM:-}" in
+	input|hid)
+		[ -n "${DEVNAME:-}" ] && rule_input "/dev/$DEVNAME"
+		# Parent "input" events often have no DEVNAME; sweep the class.
+		apply_input_all
+		;;
+	esac
+}
+
+# --- entry points -----------------------------------------------------------
+
+case "${1:-}" in
+-s|--scan)
+	apply_input_all
+	exit 0
+	;;
+-d|--daemon)
+	while true; do
+		apply_input_all
+		sleep 2
+	done
+	;;
+-h|--help)
+	cat <<'HELP'
+hotplugd — simple mdev-like hotplug for lin0
+
+  hotplugd -s     apply rules to devices already present (coldplug)
+  hotplugd -d     polling daemon (no kernel helper required)
+  hotplugd        kernel usermode helper (ACTION/SUBSYSTEM/DEVNAME in env)
+
+  echo /bin/hotplugd > /proc/sys/kernel/hotplug   # if CONFIG_UEVENT_HELPER=y
+HELP
+	exit 0
+	;;
+esac
+
+# Kernel helper invocation (or manual test with env set).
+if [ -n "${ACTION:-}" ] || [ -n "${SUBSYSTEM:-}" ]; then
+	handle_uevent
+	exit 0
+fi
+
+# Default: one-shot scan (same as -s).
+apply_input_all
